// crates/ict/src/swing.rs
use chrono::{DateTime, Utc};
use domain::{Candle, Decimal};

#[derive(Debug, Clone, PartialEq, Eq)]
pub(crate) enum SwingKind { High, Low }

#[derive(Debug, Clone)]
pub(crate) struct SwingPoint {
    pub(crate) index: usize,
    pub(crate) price: Decimal,
    pub(crate) kind: SwingKind,
    pub(crate) time: DateTime<Utc>,
}

/// N-candle pivot detection. Requires at least 2*n+1 candles.
pub(crate) fn detect_swings(candles: &[Candle], n: usize) -> Vec<SwingPoint> {
    if candles.len() < 2 * n + 1 {
        return vec![];
    }
    let mut result = Vec::new();
    for i in n..candles.len() - n {
        let left = &candles[i - n..i];
        let right = &candles[i + 1..=i + n];
        let c = &candles[i];

        let is_high = left.iter().all(|x| x.high < c.high)
            && right.iter().all(|x| x.high < c.high);
        let is_low = left.iter().all(|x| x.low > c.low)
            && right.iter().all(|x| x.low > c.low);

        if is_high {
            result.push(SwingPoint { index: i, price: c.high, kind: SwingKind::High, time: c.time });
        }
        if is_low {
            result.push(SwingPoint { index: i, price: c.low, kind: SwingKind::Low, time: c.time });
        }
    }
    result
}

## Task 3: detect.rs — 6 detection functions + 13 tests

**Files:**
- Create: `crates/ict/src/detect.rs`
- Modify: `crates/ict/src/lib.rs` (add `mod detect`)

**Key rules from spec:**
- **Bullish FVG**: `candles[i-2].high < candles[i].low` (gap up). `formed_at` = time of middle candle `candles[i-1]`
- **Bearish FVG**: `candles[i-2].low > candles[i].high` (gap down). `formed_at` = time of middle candle `candles[i-1]`
- **FVG mitigation**: scan candles after the FVG. Bullish: any candle with `low <= fvg.top` → mitigated. Bearish: any candle with `high >= fvg.bottom` → mitigated.
- **Bullish OB**: last bearish candle (`close < open`) before a swing low. `top=high`, `bottom=low`
- **Bearish OB**: last bullish candle (`close > open`) before a swing high. `top=high`, `bottom=low`
- **BOS**: candle close breaks a previous swing high (bullish BOS) or swing low (bearish BOS) in the direction of trend
- **CHoCH**: candle close breaks opposite to current trend (in uptrend, close below prev Higher Low)
- **Sweep**: candle wick exceeds swing high/low but close stays within range
- **PdArray**: uses last SwingHigh and last SwingLow from `swings`; `equilibrium = (high+low)/2`; `current_zone` from `close` of last candle
- **OTE Long**: `range = swing_high - swing_low; top = swing_high - 0.618*range; bottom = swing_high - 0.786*range`
- **OTE Short**: `range = swing_high - swing_low; top = swing_low + 0.786*range; bottom = swing_low + 0.618*range`

**BOS/CHoCH detection detail:**
- Walk through candles from index 1. Track the most recent SwingHigh and SwingLow from `swings`.
- For each candle, if `close > last_swing_high.price` → BOS Long (trend continuation)
- If `close < last_swing_low.price` → BOS Short
- CHoCH: in an uptrend (last structure was BOS Long), if `close < last_swing_low.price` → CHoCH Short (reversal)
- In a downtrend (last structure was BOS Short), if `close > last_swing_high.price` → CHoCH Long

For the tests, use a simpler approach: just check if any close breaks any swing level. Track `last_swing_high` and `last_swing_low` from `swings` as we iterate candles by time ordering.

use domain::{Candle, Decimal, Side};
use crate::swing::{SwingKind, SwingPoint};
use crate::types::{BosChoch, Fvg, LiquiditySweep, OrderBlock, Ote, PdArray, PdZone, StructureEvent};

pub(crate) fn detect_fvg(candles: &[Candle]) -> Vec<Fvg> {
    if candles.len() < 3 {
        return vec![];
    }
    let mut result = Vec::new();
    for i in 2..candles.len() {
        let prev = &candles[i - 2];
        let mid  = &candles[i - 1];
        let curr = &candles[i];

        if prev.high < curr.low {
            // bullish FVG: gap up
            let mut fvg = Fvg {
                top: curr.low,
                bottom: prev.high,
                formed_at: mid.time,
                side: Side::Long,
                mitigated: false,
            };
            // check mitigation by candles after index i
            fvg.mitigated = candles[i + 1..].iter().any(|c| c.low <= fvg.top);
            result.push(fvg);
        } else if prev.low > curr.high {
            // bearish FVG: gap down
            let mut fvg = Fvg {
                top: prev.low,
                bottom: curr.high,
                formed_at: mid.time,
                side: Side::Short,
                mitigated: false,
            };
            fvg.mitigated = candles[i + 1..].iter().any(|c| c.high >= fvg.bottom);
            result.push(fvg);
        }
    }
    result
}

pub(crate) fn detect_ob(candles: &[Candle], swings: &[SwingPoint]) -> Vec<OrderBlock> {
    let mut result = Vec::new();
    for swing in swings {
        match swing.kind {
            SwingKind::Low => {
                // bullish OB: last bearish candle before this swing low
                if let Some(ob_candle) = candles[..swing.index]
                    .iter()
                    .rev()
                    .find(|c| c.close < c.open)
                {
                    let mut ob = OrderBlock {
                        top: ob_candle.high,
                        bottom: ob_candle.low,
                        formed_at: ob_candle.time,
                        side: Side::Long,
                        mitigated: false,
                    };
                    // mitigated if any later candle closes below ob.bottom
                    ob.mitigated = candles.iter().skip(swing.index).any(|c| c.close < ob.bottom);
                    result.push(ob);
                }
            }
            SwingKind::High => {
                // bearish OB: last bullish candle before this swing high
                if let Some(ob_candle) = candles[..swing.index]
                    .iter()
                    .rev()
                    .find(|c| c.close > c.open)
                {
                    let mut ob = OrderBlock {
                        top: ob_candle.high,
                        bottom: ob_candle.low,
                        formed_at: ob_candle.time,
                        side: Side::Short,
                        mitigated: false,
                    };
                    ob.mitigated = candles.iter().skip(swing.index).any(|c| c.close > ob.top);
                    result.push(ob);
                }
            }
        }
    }
    result
}

pub(crate) fn detect_structure(candles: &[Candle], swings: &[SwingPoint]) -> Vec<BosChoch> {
    let mut result = Vec::new();
    if swings.is_empty() || candles.is_empty() {
        return result;
    }

    let last_high = swings.iter().filter(|s| s.kind == SwingKind::High).last();
    let last_low  = swings.iter().filter(|s| s.kind == SwingKind::Low).last();

    let mut last_structure_side: Option<Side> = None;

    for candle in candles {
        if let Some(sh) = last_high {
            if candle.close > sh.price {
                let kind = if last_structure_side == Some(Side::Short) {
                    StructureEvent::Choch
                } else {
                    StructureEvent::Bos
                };
                result.push(BosChoch {
                    kind,
                    level: sh.price,
                    broken_at: candle.time,
                    side: Side::Long,
                });
                last_structure_side = Some(Side::Long);
            }
        }
        if let Some(sl) = last_low {
            if candle.close < sl.price {
                let kind = if last_structure_side == Some(Side::Long) {
                    StructureEvent::Choch
                } else {
                    StructureEvent::Bos
                };
                result.push(BosChoch {
                    kind,
                    level: sl.price,
                    broken_at: candle.time,
                    side: Side::Short,
                });
                last_structure_side = Some(Side::Short);
            }
        }
    }
    result
}

pub(crate) fn detect_sweeps(candles: &[Candle], swings: &[SwingPoint]) -> Vec<LiquiditySweep> {
    let mut result = Vec::new();
    for candle in candles {
        for swing in swings {
            match swing.kind {
                SwingKind::High => {
                    // wick above high but close below → swept highs (Short setup)
                    if candle.high > swing.price && candle.close < swing.price {
                        result.push(LiquiditySweep {
                            level: swing.price,
                            swept_at: candle.time,
                            side: Side::Short,
                        });
                    }
                }
                SwingKind::Low => {
                    // wick below low but close above → swept lows (Long setup)
                    if candle.low < swing.price && candle.close > swing.price {
                        result.push(LiquiditySweep {
                            level: swing.price,
                            swept_at: candle.time,
                            side: Side::Long,
                        });
                    }
                }
            }
        }
    }
    result
}

pub(crate) fn compute_pd(candles: &[Candle], swings: &[SwingPoint]) -> Option<PdArray> {
    let range_high = swings.iter().filter(|s| s.kind == SwingKind::High).map(|s| s.price).max()?;
    let range_low  = swings.iter().filter(|s| s.kind == SwingKind::Low).map(|s| s.price).min()?;
    let last_close = candles.last()?.close;
    let two = Decimal::from(2u32);
    let equilibrium = (range_high + range_low) / two;
    let current_zone = if last_close > equilibrium {
        PdZone::Premium
    } else if last_close < equilibrium {
        PdZone::Discount
    } else {
        PdZone::Equilibrium
    };
    Some(PdArray { range_high, range_low, equilibrium, current_zone })
}

pub(crate) fn compute_ote(swings: &[SwingPoint], bias: Side) -> Option<Ote> {
    let swing_high = swings.iter().filter(|s| s.kind == SwingKind::High).map(|s| s.price).max()?;
    let swing_low  = swings.iter().filter(|s| s.kind == SwingKind::Low).map(|s| s.price).min()?;
    let range = swing_high - swing_low;
    let fib618: Decimal = FIB_618.parse().unwrap();
    let fib786: Decimal = FIB_786.parse().unwrap();
    let (top, bottom) = match bias {
        Side::Long  => (swing_high - fib618 * range, swing_high - fib786 * range),
        Side::Short => (swing_low + fib786 * range,  swing_low + fib618 * range),
    };
    Some(Ote { top, bottom, side: bias })
}

fn check_confluence(
    candles: &[Candle],
    fvgs: &[Fvg],
    order_blocks: &[OrderBlock],
    structure: &[BosChoch],
    sweeps: &[LiquiditySweep],
    pd_array: &Option<PdArray>,
    ote: &Option<Ote>,
    swings: &[crate::swing::SwingPoint],
) -> Option<TradeSignal> {
    use crate::swing::SwingKind;

    // 1. Need a BOS/CHoCH to establish bias
    let last_structure = structure.last()?;
    let bias = last_structure.side.clone();

    // 2. PD zone must match bias
    let pd = pd_array.as_ref()?;
    let zone_ok = match bias {
        Side::Long  => pd.current_zone == PdZone::Discount,
        Side::Short => pd.current_zone == PdZone::Premium,
    };
    if !zone_ok {
        return None;
    }

    // 3. Last close must be inside OTE zone
    let ote = ote.as_ref()?;
    let last_close = candles.last()?.close;
    if last_close < ote.bottom || last_close > ote.top {
        return None;
    }

    // 4. Find overlapping unmitigated structure (OB preferred over FVG)
    let overlapping_ob = order_blocks.iter()
        .filter(|o| !o.mitigated && o.side == bias && o.top > ote.bottom && o.bottom < ote.top)
        .last();

    let overlapping_fvg = fvgs.iter()
        .filter(|f| !f.mitigated && f.side == bias && f.top > ote.bottom && f.bottom < ote.top)
        .last();

    let (entry_top, entry_bottom, sl) = if let Some(ob) = overlapping_ob {
        let sl = match bias { Side::Long => ob.bottom, Side::Short => ob.top };
        (ob.top, ob.bottom, sl)
    } else if let Some(fvg) = overlapping_fvg {
        let sl = match bias { Side::Long => fvg.bottom, Side::Short => fvg.top };
        (fvg.top, fvg.bottom, sl)
    } else {
        return None;
    };

    let two = Decimal::from(2u32);
    let entry = (entry_top + entry_bottom) / two;

    let tp = match bias {
        Side::Long  => swings.iter().filter(|s| s.kind == SwingKind::High).map(|s| s.price).max()?,
        Side::Short => swings.iter().filter(|s| s.kind == SwingKind::Low).map(|s| s.price).min()?,
    };

    // sweep_present: any sweep in last 5 candles
    let last_5_start = candles.len().saturating_sub(5);
    let cutoff_time = candles[last_5_start].time;
    let sweep_present = sweeps.iter().any(|s| s.swept_at >= cutoff_time);

    let confluence = ConfluenceFlags {
        has_bos_choch: true,
        in_pd_zone: true,
        ob_in_ote: overlapping_ob.is_some(),
        fvg_in_ote: overlapping_fvg.is_some(),
        sweep_present,
    };

    Some(TradeSignal { side: bias, entry, sl, tp, confluence })
}
